import { _electron } from "@playwright/test";
import { mkdtemp, rm, writeFile } from "node:fs/promises";
import { tmpdir } from "node:os";
import path from "node:path";
import { test as base, expect, type Page } from "./fixtures";

export { expect, type Page };

export const test = base.extend({
  page: async ({ e2eWorker }, provide, testInfo) => {
    void e2eWorker;
    const directory = await mkdtemp(path.join(tmpdir(), "paseo-9437-electron-"));
    const endpoint = `127.0.0.1:${process.env.E2E_DAEMON_PORT}`;
    const env = Object.fromEntries(
      Object.entries(process.env).filter(([, value]) => typeof value === "string"),
    );
    delete env.ELECTRON_RUN_AS_NODE;
    const launcher = path.join(directory, "launch.cjs");
    await writeFile(launcher, `process.argv = [process.argv[0]]; require(${JSON.stringify(path.resolve(__dirname, "../../../desktop/dist/main.js"))});`);
    const app = await _electron.launch({
      executablePath: process.env.PASEO_E2E_ELECTRON_BINARY ?? require("electron"),
      args: [launcher],
      cwd: path.resolve(__dirname, "../../../.."),
      env: {
        ...env,
        PASEO_HOME: process.env.E2E_PASEO_HOME,
        PASEO_LISTEN: endpoint,
        PASEO_DAEMON_ENDPOINT: endpoint,
        PASEO_ELECTRON_USER_DATA_DIR: directory,
        PASEO_NODE_ENV: "development",
        EXPO_DEV_URL: `http://localhost:${process.env.E2E_METRO_PORT}`,
        PASEO_LOCAL_SPEECH_AUTO_DOWNLOAD: "0",
      },
      recordVideo: { dir: testInfo.outputPath("electron-video"), size: { width: 1280, height: 800 } },
      timeout: 60_000,
    });
    try {
      const page = await app.firstWindow();
      await page.waitForURL((url) => url.origin === `http://localhost:${process.env.E2E_METRO_PORT}`, { waitUntil: "domcontentloaded" });
      await page.waitForFunction(() => document.getElementById("root")?.childElementCount > 0);
      await page.setViewportSize({ width: 1280, height: 800 });
      const goto = page.goto.bind(page);
      page.goto = (url, options) => goto(new URL(url, `http://localhost:${process.env.E2E_METRO_PORT}`).href, options);
      await provide(page);
    } finally {
      await app.close();
      await rm(directory, { recursive: true, force: true });
    }
  },
});
